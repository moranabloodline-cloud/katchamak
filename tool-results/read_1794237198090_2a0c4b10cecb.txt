   110→interface LegacyCodexEntry {
   111→  id: string;

   113→  content: string;

   121→interface LegacyCodexCategory {
   122→  id: string;
   123→  name: string;
   124→  children: LegacyCodexCategory[];
   125→  entries: LegacyCodexEntry[];
   126→}
   127→
   128→type LegacyNovel = Novel & { codex?: LegacyCodexCategory[] };
   129→
   130→const LEGACY_CAT_ICONS: [RegExp, string][] = [
   131→  [/^char/i, "users"],
   132→  [/^loc/i, "map-pin"],
   133→  [/^thing/i, "diamond"],
   134→  [/^lore/i, "scroll"],
   135→];

   137→/* turn an old codex category (with its entries) into a plain note subtree */
   138→function legacyCatToNode(cat: LegacyCodexCategory): NoteNode {
   139→  const now = Date.now();
   140→  const icon = LEGACY_CAT_ICONS.find(([re]) => re.test(cat.name))?.[1] ?? "folder";
   141→  return {
   142→    id: uid(),
   143→    title: cat.name,
   144→    kind: "folder",
   145→    synopsis: "",
   146→    content: "",
   147→    notes: "",
   148→    status: "",
   149→    label: "",
   150→    icon,
   151→    createdAt: now,
   152→    updatedAt: now,
   153→    children: [
   154→      ...cat.children.map(legacyCatToNode),
   155→      ...cat.entries.map((e) => ({
   156→        id: uid(),
   157→        title: e.title,
   158→        kind: "doc" as const,
   159→        synopsis: "",
   160→        content: e.content ?? "",
   161→        notes: "",
   162→        status: e.status ?? "",
   163→        label: e.entryType ?? "",
   164→        createdAt: e.createdAt ?? now,
   165→        updatedAt: e.updatedAt ?? now,
   166→        children: [] as NoteNode[],
   167→      })),
   168→    ],
   169→  };
   170→}
   171→
   172→/* make sure a novel has every section (older backups may miss keys) and convert
   173→   its legacy codex array into plain notes in the Codex section */
   174→function normalizeNovelSections(n: LegacyNovel): void {
   175→  const secs: Record<string, NoteNode[] | undefined> = { ...(n.sections ?? {}) };
   176→  for (const k of SECTION_KEYS) {
   177→    if (!Array.isArray(secs[k])) secs[k] = [];
   178→  }
   179→  if (secs.codex!.length === 0 && Array.isArray(n.codex) && n.codex.length > 0) {
   180→    secs.codex = n.codex.map(legacyCatToNode);
   181→  }
   182→  n.sections = secs as Novel["sections"];
   183→  delete n.codex;
   184→}

   186→/* ---------- immutable helpers ---------- */
   187→
   188→function mapNodes(nodes: NoteNode[], fn: (n: NoteNode) => NoteNode): NoteNode[] {
   189→  return nodes.map((n) => ({ ...fn(n), children: mapNodes(n.children, fn) }));
   190→}
   191→
   192→function removeFromNodes(
   193→  nodes: NoteNode[],
   194→  id: string
   195→): { nodes: NoteNode[]; removed: NoteNode | null; parentId: string | null } {
   196→  const found: { removed: NoteNode | null; parentId: string | null } = {
   197→    removed: null,
   198→    parentId: null,
   199→  };
   200→  const walk = (list: NoteNode[], pid: string | null): NoteNode[] => {
   201→    const out: NoteNode[] = [];
   202→    for (const n of list) {
   203→      if (n.id === id) {
   204→        found.removed = n;
   205→        found.parentId = pid;
   206→        continue;
   207→      }
   208→      out.push({ ...n, children: walk(n.children, n.id) });
   209→    }
   210→    return out;
   211→  };
   212→  const nodes2 = walk(nodes, null);
   213→  return { nodes: nodes2, removed: found.removed, parentId: found.parentId };
   214→}

   216→function touch(novel: Novel): Novel {
   217→  return { ...novel, updatedAt: Date.now() };
   218→}

   220→function makeNode(kind: NoteNode["kind"], title: string): NoteNode {
   221→  const now = Date.now();
   222→  return {
   223→    id: uid(),
   224→    title,
   225→    kind,
   226→    synopsis: "",
   227→    content: "",
   228→    notes: "",
   229→    status: "",
   230→    label: "",
   231→    createdAt: now,
   232→    updatedAt: now,
   233→    children: [],
   234→  };
   235→}

   407→      addNode: (section, parentId, kind, title) => {
   408→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   409→        if (!novel) return null;
   410→        const roots = novel.sections[section];
   411→        let created: NoteNode | null = null;

   413→        /* free hierarchy — every new note is the same thing; the user names it

   416→        if (parentId === null) {
   417→          created = n;
   418→          roots.push(n);
   419→        } else {

   421→          if (!parent) return null;
   422→          created = n;
   423→          parent.children.push(n);
   424→          parent.collapsed = false;

   428→          renamingNodeId: created?.id ?? null,

   460→        set((s) => ({
   461→          selectedNodeId:

   463→            findNode([removedNode], s.selectedNodeId ?? "")
   464→              ? null
   465→              : s.selectedNodeId,
   466→          renamingNodeId:
   467→            s.renamingNodeId &&
   468→            (s.renamingNodeId === id || findNode([removedNode], s.renamingNodeId))
   469→              ? null
   470→              : s.renamingNodeId,
   471→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   472→        }));
   473→      },
   474→
   475→      moveNode: (section, id, dir) => {
   476→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   477→        if (!novel) return;
   478→        const roots = novel.sections[section];
   479→
   480→        const swap = (list: NoteNode[]): boolean => {
   481→          const i = list.findIndex((n) => n.id === id);
   482→          if (i >= 0) {
   483→            const j = dir === "up" ? i - 1 : i + 1;
   484→            if (j < 0 || j >= list.length) return false;
   485→            const copy = [...list];
   486→            [copy[i], copy[j]] = [copy[j], copy[i]];
   487→            list.length = 0;
   488→            list.push(...copy);
   489→            return true;
   490→          }
   491→          return list.some((n) => swap(n.children));
   492→        };
   493→        swap(roots);
   494→        set((s) => ({
   495→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   496→        }));
   497→      },

   499→      /* Move — the note becomes the last child of the destination (or a section root) */
   500→      moveNodeTo: (fromSection, id, destSection, destParentId, destIndex) => {
   501→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   502→        if (!novel) return;
   503→        const fromRoots = novel.sections[fromSection];
   504→        const moving = findNode(fromRoots, id);
   505→        if (!moving) return;
   506→        /* never move a note into itself or one of its own sub-pages */
   507→        if (
   508→          destSection === fromSection &&
   509→          destParentId &&
   510→          (destParentId === id || findNode([moving], destParentId))
   511→        )
   512→          return;
   513→        const { nodes, removed } = removeFromNodes(fromRoots, id);
   514→        if (!removed) return;
   515→        novel.sections = { ...novel.sections, [fromSection]: nodes };
   516→        /* destIndex (optional): insertion position in the post-removal list —
   517→           the Corkboard computes it against the array without the dragged note */
   518→        const spliceOrPush = (list: NoteNode[]) => {
   519→          if (typeof destIndex === "number") {
   520→            const idx = Math.max(0, Math.min(destIndex, list.length));
   521→            list.splice(idx, 0, removed);
   522→          } else {
   523→            list.push(removed);
   524→          }
   525→        };
   526→        if (destParentId) {
   527→          const parent = findNode(novel.sections[destSection], destParentId);
   528→          if (parent) {
   529→            spliceOrPush(parent.children);
   530→            parent.collapsed = false;
   531→          } else {
   532→            spliceOrPush(novel.sections[destSection]);
   533→          }
   534→        } else {
   535→          spliceOrPush(novel.sections[destSection]);
   536→        }
   537→        set((s) => ({
   538→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   539→        }));
   540→      },
   541→
   542→      /* deep-copy the note (with all sub-pages) right below the original */
   543→      duplicateNode: (section, id) => {
   544→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   545→        if (!novel) return;
   546→        const roots = novel.sections[section];
   547→        if (!findNode(roots, id)) return;
   548→        const deep = (n: NoteNode): NoteNode => ({
   549→          ...n,
   550→          id: uid(),
   551→          createdAt: Date.now(),
   552→          updatedAt: Date.now(),
   553→          children: n.children.map(deep),
   554→        });
   555→        const insertAfter = (list: NoteNode[]): boolean => {
   556→          const i = list.findIndex((n) => n.id === id);
   557→          if (i >= 0) {
   558→            list.splice(i + 1, 0, deep(list[i]));
   559→            return true;
   560→          }
   561→          return list.some((n) => insertAfter(n.children));
   562→        };
   563→        insertAfter(roots);
   564→        set((s) => ({
   565→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   566→        }));
   567→      },
   568→
   569→      /* sort sub-pages of a note — title A→Z / Z→A, oldest or newest first */
   570→      sortChildren: (section, id, mode) => {
   571→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   572→        if (!novel) return;
   573→        const roots = novel.sections[section];
   574→        const target = findNode(roots, id);
   575→        const list = target ? target.children : roots;
   576→        if (!target || list.length === 0) return;
   577→        const byTitle = (a: NoteNode, b: NoteNode) =>
   578→          a.title.localeCompare(b.title, undefined, {
   579→            sensitivity: "base",
   580→            numeric: true,
   581→          });
   582→        list.sort((a, b) => {
   583→          switch (mode) {
   584→            case "title-desc":
   585→              return byTitle(b, a);
   586→            case "created-asc":
   587→              return a.createdAt - b.createdAt;
   588→            case "created-desc":
   589→              return b.createdAt - a.createdAt;
   590→            default:
   591→              return byTitle(a, b);
   592→          }
   593→        });
   594→        set((s) => ({
   595→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   596→        }));
   597→      },
   598→
   599→      /* ---------- trash ---------- */
   600→
   601→      restoreFromTrash: (trashId) => {
   602→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   603→        if (!novel) return;
   604→        const item = novel.trash.find((t) => t.id === trashId);
   605→        if (!item) return;
   606→        if (item.parentId) {
   607→          const parent = findNode(novel.sections[item.section], item.parentId);
   608→          if (parent) {
   609→            parent.children.push(item.node);
   610→            parent.collapsed = false;
   611→          } else {
   612→            novel.sections[item.section].push(item.node);
   613→          }
   614→        } else {
   615→          novel.sections[item.section].push(item.node);
   616→        }
   617→        novel.trash = novel.trash.filter((t) => t.id !== trashId);
   618→        set((s) => ({
   619→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   620→        }));
   621→      },
   622→
   623→      deleteForever: (trashId) => {
   624→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   625→        if (!novel) return;
   626→        novel.trash = novel.trash.filter((t) => t.id !== trashId);
   627→        set((s) => ({
   628→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   629→        }));
   630→      },
   631→
   632→      emptyTrash: () => {
   633→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   634→        if (!novel) return;
   635→        novel.trash = [];
   636→        set((s) => ({
   637→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   638→        }));
   639→      },
   640→    }),
   641→    {
   642→      name: STORAGE_KEY,
   643→      version: 5,
   644→      storage: createJSONStorage(() => localStorage),
   645→      /* v0 → v1: original-style statuses, free-text labels instead of label ids
   646→         v2 → v3: every note gets its icon + colour (like the original app —
   647→         acts and chapters as solid coloured folders)
   648→         v3 → v4: everything is a NOTE on every tab — the old Codex
   649→         categories/entries become plain notes in the Codex section
   650→         v4 → v5: the Grid view becomes the Corkboard (Scrivener index cards) */
   651→      migrate: (persisted) => {
   652→        const s = persisted as KachamakState;
   653→        if ((s as { view?: string }).view === "grid") s.view = "corkboard";
   654→        const OLD_DEFAULT_STATUSES = [
   655→          "To Do",
   656→          "First Draft",
   657→          "Second Draft",
   658→          "Revised",
   659→          "Final",
   660→        ];
   661→        const RENAMES: Record<string, string> = {
   662→          "Second Draft": "Revised Draft",
   663→          Revised: "Revised Draft",
   664→          Final: "Final Draft",
   665→        };
   666→        for (const n of (s.novels ?? []) as LegacyNovel[]) {
   667→          let upgraded = false;
   668→          if ((n.statuses ?? []).join("|") === OLD_DEFAULT_STATUSES.join("|")) {
   669→            n.statuses = [...DEFAULT_STATUSES];
   670→            upgraded = true;
   671→          }
   672→          const idToName = new Map((n.labels ?? []).map((l) => [l.id, l.name]));
   673→          const nameToColor = new Map((n.labels ?? []).map((l) => [l.name, l.color]));
   674→          /* act palette like the original screenshot: gold / green / purple … */
   675→          const ACT_PALETTE = [
   676→            { act: "#e3c567", chapter: "#c69c6d" },
   677→            { act: "#8ab661", chapter: "#699bdd" },
   678→            { act: "#a371f7", chapter: "#c084fc" },
   679→          ];
   680→          let actIndex = -1;
   681→          const walk = (nodes: NoteNode[], chapterColor?: string) => {
   682→            for (const nd of nodes) {
   683→              if (nd.label && idToName.has(nd.label)) nd.label = idToName.get(nd.label)!;
   684→              if (!nd.color && nd.label && nameToColor.has(nd.label)) {
   685→                nd.color = nameToColor.get(nd.label)!;
   686→              }
   687→              if (upgraded && RENAMES[nd.status]) nd.status = RENAMES[nd.status];
   688→              /* give every container-like note the solid folder icon + colours */
   689→              if (!nd.icon && (nd.kind === "act" || nd.kind === "chapter" || nd.kind === "folder")) {
   690→                nd.icon = "folder";
   691→              }
   692→              let chColor: string | undefined;
   693→              if (!nd.color && nd.kind === "act") {
   694→                actIndex += 1;
   695→                const pal = ACT_PALETTE[actIndex % ACT_PALETTE.length];
   696→                nd.color = pal.act;
   697→                chColor = pal.chapter;
   698→              }
   699→              if (!nd.color && nd.kind === "chapter" && chapterColor) {
   700→                nd.color = chapterColor;
   701→              }
   702→              walk(nd.children, nd.kind === "act" ? chColor : undefined);
   703→            }
   704→          };
   705→          normalizeNovelSections(n);
   706→          for (const roots of Object.values(n.sections)) {
   707→            actIndex = -1;
   708→            walk(roots);
   709→          }
   710→          for (const t of n.trash ?? []) walk([t.node]);
   711→        }
   712→        return s;
   713→      },
   714→      onRehydrateStorage: () => (state) => {
   715→        // first run: seed with the demo novels
   716→        if (state && state.novels.length === 0) {
   717→          state.novels = seedNovels();
   718→        }
   719→      },
   720→    }
   721→  )
   722→);
