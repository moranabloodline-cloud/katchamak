    93→    93→function mapNodes(nodes: NoteNode[], fn: (n: NoteNode) => NoteNode): NoteNode[] {
    94→    94→  return nodes.map((n) => ({ ...fn(n), children: mapNodes(n.children, fn) }));
    95→    95→}

    97→    97→function removeFromNodes(
    98→    98→  nodes: NoteNode[],
    99→    99→  id: string
   100→   100→): { nodes: NoteNode[]; removed: NoteNode | null; parentId: string | null } {
   101→   101→  const found: { removed: NoteNode | null; parentId: string | null } = {
   102→   102→    removed: null,
   103→   103→    parentId: null,
   104→   104→  };
   105→   105→  const walk = (list: NoteNode[], pid: string | null): NoteNode[] => {
   106→   106→    const out: NoteNode[] = [];
   107→   107→    for (const n of list) {
   108→   108→      if (n.id === id) {
   109→   109→        found.removed = n;
   110→   110→        found.parentId = pid;
   111→   111→        continue;
   112→   112→      }
   113→   113→      out.push({ ...n, children: walk(n.children, n.id) });
   114→   114→    }
   115→   115→    return out;
   116→   116→  };
   117→   117→  const nodes2 = walk(nodes, null);
   118→   118→  return { nodes: nodes2, removed: found.removed, parentId: found.parentId };
   119→   119→}

   121→   121→function mapCategories(
   122→   122→  cats: CodexCategory[],
   123→   123→  fn: (c: CodexCategory) => CodexCategory
   124→   124→): CodexCategory[] {
   125→   125→  return cats.map((c) => ({ ...fn(c), children: mapCategories(c.children, fn) }));
   126→   126→}
   127→   127→
   128→   128→function removeFromCategories(
   129→   129→  cats: CodexCategory[],
   130→   130→  id: string
   131→   131→): { cats: CodexCategory[]; removed: CodexCategory | null } {
   132→   132→  const found: { removed: CodexCategory | null } = { removed: null };
   133→   133→  const walk = (list: CodexCategory[]): CodexCategory[] => {
   134→   134→    const out: CodexCategory[] = [];
   135→   135→    for (const c of list) {
   136→   136→      if (c.id === id) {
   137→   137→        found.removed = c;
   138→   138→        continue;
   139→   139→      }
   140→   140→      out.push({ ...c, children: walk(c.children) });
   141→   141→    }
   142→   142→    return out;
   143→   143→  };
   144→   144→  const cats2 = walk(cats);
   145→   145→  return { cats: cats2, removed: found.removed };
   146→   146→}

   148→   148→function findCategory(
   149→   149→  cats: CodexCategory[],
   150→   150→  id: string
   151→   151→): CodexCategory | null {
   152→   152→  for (const c of cats) {
   153→   153→    if (c.id === id) return c;
   154→   154→    const hit = findCategory(c.children, id);
   155→   155→    if (hit) return hit;
   156→   156→  }
   157→   157→  return null;
   158→   158→}

   411→   411→            if (j < 0 || j >= list.length) return false;
   412→   412→            const copy = [...list];
   413→   413→            [copy[i], copy[j]] = [copy[j], copy[i]];
   414→   414→            list.length = 0;
   415→   415→            list.push(...copy);
   416→   416→            return true;
   417→   417→          }

   419→   419→        };

   433→   433→          ...n,
   434→   434→          id: uid(),
   435→   435→          createdAt: Date.now(),
   436→   436→          updatedAt: Date.now(),
   437→   437→          children: n.children.map(deep),
   438→   438→        });

   440→   440→          const i = list.findIndex((n) => n.id === id);
   441→   441→          if (i >= 0) {
   442→   442→            list.splice(i + 1, 0, deep(list[i]));
   443→   443→            return true;
   444→   444→          }
   445→   445→          return list.some((n) => insertAfter(n.children));
   446→   446→        };
   447→   447→        insertAfter(roots);
   448→   448→        set((s) => ({
   449→   449→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   450→   450→        }));
   451→   451→      },
   452→   452→
   453→   453→      /* sort sub-pages of a note alphabetically (A→Z) */

   455→   455→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   456→   456→        if (!novel) return;

   459→   459→        const list = target ? target.children : roots;
   460→   460→        if (!target || list.length === 0) return;
   461→   461→        list.sort((a, b) =>
   462→   462→          a.title.localeCompare(b.title, undefined, {
   463→   463→            sensitivity: "base",
   464→   464→            numeric: true,
   465→   465→          })
   466→   466→        );
   467→   467→        set((s) => ({
   468→   468→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   469→   469→        }));
   470→   470→      },
   471→   471→
   472→   472→      /* ---------- trash ---------- */
   473→   473→
   474→   474→      restoreFromTrash: (trashId) => {
   475→   475→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   476→   476→        if (!novel) return;
   477→   477→        const item = novel.trash.find((t) => t.id === trashId);

   642→   642→    }),

   650→   650→      migrate: (persisted) => {
