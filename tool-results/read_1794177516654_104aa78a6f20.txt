    91→/* ---------- immutable helpers ---------- */
    92→
    93→function mapNodes(nodes: NoteNode[], fn: (n: NoteNode) => NoteNode): NoteNode[] {
    94→  return nodes.map((n) => ({ ...fn(n), children: mapNodes(n.children, fn) }));
    95→}
    96→
    97→function removeFromNodes(
    98→  nodes: NoteNode[],
    99→  id: string
   100→): { nodes: NoteNode[]; removed: NoteNode | null; parentId: string | null } {
   101→  const found: { removed: NoteNode | null; parentId: string | null } = {
   102→    removed: null,
   103→    parentId: null,
   104→  };
   105→  const walk = (list: NoteNode[], pid: string | null): NoteNode[] => {
   106→    const out: NoteNode[] = [];
   107→    for (const n of list) {
   108→      if (n.id === id) {
   109→        found.removed = n;
   110→        found.parentId = pid;
   111→        continue;
   112→      }
   113→      out.push({ ...n, children: walk(n.children, n.id) });
   114→    }
   115→    return out;
   116→  };
   117→  const nodes2 = walk(nodes, null);
   118→  return { nodes: nodes2, removed: found.removed, parentId: found.parentId };
   119→}
   120→
   121→function mapCategories(
   122→  cats: CodexCategory[],
   123→  fn: (c: CodexCategory) => CodexCategory
   124→): CodexCategory[] {
   125→  return cats.map((c) => ({ ...fn(c), children: mapCategories(c.children, fn) }));
   126→}
   127→
   128→function removeFromCategories(
   129→  cats: CodexCategory[],
   130→  id: string
   131→): { cats: CodexCategory[]; removed: CodexCategory | null } {
   132→  const found: { removed: CodexCategory | null } = { removed: null };
   133→  const walk = (list: CodexCategory[]): CodexCategory[] => {
   134→    const out: CodexCategory[] = [];
   135→    for (const c of list) {
   136→      if (c.id === id) {
   137→        found.removed = c;
   138→        continue;
   139→      }
   140→      out.push({ ...c, children: walk(c.children) });
   141→    }
   142→    return out;
   143→  };
   144→  const cats2 = walk(cats);
   145→  return { cats: cats2, removed: found.removed };
   146→}
   147→
   148→function findCategory(
   149→  cats: CodexCategory[],
   150→  id: string
   151→): CodexCategory | null {
   152→  for (const c of cats) {
   153→    if (c.id === id) return c;
   154→    const hit = findCategory(c.children, id);
   155→    if (hit) return hit;
   156→  }
   157→  return null;
   158→}

   160→function touch(novel: Novel): Novel {
   161→  return { ...novel, updatedAt: Date.now() };
   162→}

   164→function makeNode(kind: NoteNode["kind"], title: string): NoteNode {
   165→  const now = Date.now();
   166→  return {
   167→    id: uid(),
   168→    title,
   169→    kind,
   170→    synopsis: "",
   171→    content: "",
   172→    notes: "",
   173→    status: "",
   174→    label: "",
   175→    createdAt: now,
   176→    updatedAt: now,
   177→    children: [],
   178→  };
   179→}

   313→      setView: (v) => set({ view: v }),
   314→
   315→      select: (id) => set({ selectedNodeId: id }),
   316→
   317→      selectCodex: (id) => set({ selectedCodexId: id }),
   318→
   319→      renameNovel: (title) =>
   320→        set((s) => ({
   321→          novels: s.novels.map((n) =>
   322→            n.id === s.activeNovelId ? touch({ ...n, title: title.trim() || n.title }) : n
   323→          ),
   324→        })),
   325→
   326→      setDefaultAuthor: (a) => set({ defaultAuthor: a }),
   327→
   328→      /* ---------- notes ---------- */
   329→
   330→      addNode: (section, parentId, kind, title) => {
   331→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   332→        if (!novel) return null;
   333→        const roots = novel.sections[section];
   334→        let created: NoteNode | null = null;
   335→
   336→        const fallbackTitle =
   337→          kind === "act"
   338→            ? "New Act"
   339→            : kind === "chapter"
   340→              ? "New Chapter"
   341→              : kind === "scene"
   342→                ? "New Scene"
   343→                : kind === "folder"
   344→                  ? "New Folder"
   345→                  : "New Note";
   346→
   347→        if (parentId === null) {
   348→          const n = makeNode(kind, title ?? fallbackTitle);
   349→          created = n;
   350→          roots.push(n);
   351→        } else {
   352→          const parent = findNode(roots, parentId);
   353→          if (!parent) return null;
   354→          const n = makeNode(kind, title ?? fallbackTitle);
   355→          created = n;
   356→          parent.children.push(n);
   357→          parent.collapsed = false;
   358→        }
   359→        set((s) => ({
   360→          selectedNodeId: created?.id ?? null,
   361→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   362→        }));
   363→        return created?.id ?? null;
   364→      },
   365→
   366→      updateNode: (section, id, patch) => {
   367→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   368→        if (!novel) return;
   369→        const roots = novel.sections[section];
   370→        const target = findNode(roots, id);
   371→        if (!target) return;
   372→        Object.assign(target, patch, { updatedAt: Date.now() });
   373→        set((s) => ({
   374→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   375→        }));
   376→      },
   377→
   378→      deleteNode: (section, id) => {
   379→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   380→        if (!novel) return;
   381→        const { nodes, removed, parentId } = removeFromNodes(
   382→          novel.sections[section],
   383→          id
   384→        );
   385→        if (!removed) return;
   386→        novel.sections = { ...novel.sections, [section]: nodes };
   387→        novel.trash = [
   388→          { id: uid(), parentId, section, node: removed, deletedAt: Date.now() },
   389→          ...novel.trash,
   390→        ];
   391→        const removedNode: NoteNode = removed;
   392→        set((s) => ({
   393→          selectedNodeId:
   394→            s.selectedNodeId === id ||
   395→            findNode([removedNode], s.selectedNodeId ?? "")
   396→              ? null
   397→              : s.selectedNodeId,
   398→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   399→        }));
   400→      },
   401→
   402→      moveNode: (section, id, dir) => {
   403→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   404→        if (!novel) return;
   405→        const roots = novel.sections[section];
   406→
   407→        const swap = (list: NoteNode[]): boolean => {
   408→          const i = list.findIndex((n) => n.id === id);
   409→          if (i >= 0) {
   410→            const j = dir === "up" ? i - 1 : i + 1;
   411→            if (j < 0 || j >= list.length) return false;
   412→            const copy = [...list];
   413→            [copy[i], copy[j]] = [copy[j], copy[i]];
   414→            list.length = 0;
   415→            list.push(...copy);
   416→            return true;
   417→          }
   418→          return list.some((n) => swap(n.children));
   419→        };
   420→        swap(roots);
   421→        set((s) => ({
   422→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   423→        }));
   424→      },
   425→
   426→      /* deep-copy the note (with all sub-pages) right below the original */
   427→      duplicateNode: (section, id) => {
   428→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   429→        if (!novel) return;
   430→        const roots = novel.sections[section];
   431→        if (!findNode(roots, id)) return;
   432→        const deep = (n: NoteNode): NoteNode => ({
   433→          ...n,
   434→          id: uid(),
   435→          createdAt: Date.now(),
   436→          updatedAt: Date.now(),
   437→          children: n.children.map(deep),
   438→        });
   439→        const insertAfter = (list: NoteNode[]): boolean => {
   440→          const i = list.findIndex((n) => n.id === id);
   441→          if (i >= 0) {
   442→            list.splice(i + 1, 0, deep(list[i]));
   443→            return true;
   444→          }
   445→          return list.some((n) => insertAfter(n.children));
   446→        };
   447→        insertAfter(roots);
   448→        set((s) => ({
   449→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   450→        }));
   451→      },
   452→
   453→      /* sort sub-pages of a note alphabetically (A→Z) */
   454→      sortChildren: (section, id) => {
   455→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   456→        if (!novel) return;
   457→        const roots = novel.sections[section];
   458→        const target = findNode(roots, id);
   459→        const list = target ? target.children : roots;
   460→        if (!target || list.length === 0) return;
   461→        list.sort((a, b) =>
   462→          a.title.localeCompare(b.title, undefined, {
   463→            sensitivity: "base",
   464→            numeric: true,
   465→          })
   466→        );
   467→        set((s) => ({
   468→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   469→        }));
   470→      },
   471→
   472→      /* ---------- trash ---------- */
   473→
   474→      restoreFromTrash: (trashId) => {
   475→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   476→        if (!novel) return;
   477→        const item = novel.trash.find((t) => t.id === trashId);
   478→        if (!item) return;
   479→        if (item.parentId) {
   480→          const parent = findNode(novel.sections[item.section], item.parentId);
   481→          if (parent) {
   482→            parent.children.push(item.node);
   483→            parent.collapsed = false;
   484→          } else {
   485→            novel.sections[item.section].push(item.node);
   486→          }
   487→        } else {
   488→          novel.sections[item.section].push(item.node);
   489→        }
   490→        novel.trash = novel.trash.filter((t) => t.id !== trashId);
   491→        set((s) => ({
   492→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   493→        }));
   494→      },
   495→
   496→      deleteForever: (trashId) => {
   497→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   498→        if (!novel) return;
   499→        novel.trash = novel.trash.filter((t) => t.id !== trashId);
   500→        set((s) => ({
   501→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   502→        }));
   503→      },
   504→
   505→      emptyTrash: () => {
   506→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   507→        if (!novel) return;
   508→        novel.trash = [];
   509→        set((s) => ({
   510→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   511→        }));
   512→      },
   513→
   514→      /* ---------- codex ---------- */
   515→
   516→      addCategory: (parentId, name) => {
   517→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   518→        if (!novel) return;
   519→        const cat: CodexCategory = {
   520→          id: uid(),
   521→          name: name.trim() || "New Folder",
   522→          children: [],
   523→          entries: [],
   524→        };
   525→        if (parentId === null) {
   526→          novel.codex.push(cat);
   527→        } else {
   528→          const parent = findCategory(novel.codex, parentId);
   529→          if (parent) {
   530→            parent.children.push(cat);
   531→          } else {
   532→            novel.codex.push(cat);
   533→          }
   534→        }
   535→        set((s) => ({
   536→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   537→        }));
   538→      },
   539→
   540→      renameCategory: (id, name) => {
   541→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   542→        if (!novel) return;
   543→        const cat = findCategory(novel.codex, id);
   544→        if (cat) cat.name = name.trim() || cat.name;
   545→        set((s) => ({
   546→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   547→        }));
   548→      },
   549→
   550→      deleteCategory: (id) => {
   551→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   552→        if (!novel) return;
   553→        const { cats, removed } = removeFromCategories(novel.codex, id);
   554→        if (!removed) return;
   555→        novel.codex = cats;
   556→        // deleted entries' ids no longer selected
   557→        const entryIds = new Set<string>();
   558→        const collect = (c: CodexCategory) => {
   559→          c.entries.forEach((e) => entryIds.add(e.id));
   560→          c.children.forEach(collect);
   561→        };
   562→        collect(removed);
   563→        set((s) => ({
   564→          selectedCodexId: s.selectedCodexId && entryIds.has(s.selectedCodexId) ? null : s.selectedCodexId,
   565→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   566→        }));
   567→      },
   568→
   569→      addEntry: (categoryId, title) => {
   570→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   571→        if (!novel) return null;
   572→        const cat = findCategory(novel.codex, categoryId);
   573→        if (!cat) return null;
   574→        const e: CodexEntry = {
   575→          id: uid(),
   576→          title: title.trim() || "New Entry",
   577→          content: "",
   578→          aliases: [],
   579→          entryType: "Human",
   580→          status: "To Do",
   581→          createdAt: Date.now(),
   582→          updatedAt: Date.now(),
   583→        };
   584→        cat.entries.push(e);
   585→        set((s) => ({
   586→          selectedCodexId: e.id,
   587→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   588→        }));
   589→        return e.id;
   590→      },
   591→
   592→      updateEntry: (id, patch) => {
   593→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   594→        if (!novel) return;
   595→        const found: { hit: boolean } = { hit: false };
   596→        novel.codex = mapCategories(novel.codex, (c) => ({
   597→          ...c,
   598→          entries: c.entries.map((e) => {
   599→            if (e.id === id) {
   600→              found.hit = true;
   601→              return { ...e, ...patch, updatedAt: Date.now() };
   602→            }
   603→            return e;
   604→          }),
   605→        }));
   606→        if (found.hit) {
   607→          set((s) => ({
   608→            novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   609→          }));
   610→        }
   611→      },
   612→
   613→      deleteEntry: (id) => {
   614→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   615→        if (!novel) return;
   616→        mapCategories(novel.codex, (c) => {
   617→          c.entries = c.entries.filter((e) => e.id !== id);
   618→          return c;
   619→        });
   620→        set((s) => ({
   621→          selectedCodexId: s.selectedCodexId === id ? null : s.selectedCodexId,
   622→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   623→        }));
   624→      },
   625→
   626→      findEntryCategory: (entryId) => {
   627→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   628→        if (!novel) return null;
   629→        let found: CodexCategory | null = null;
   630→        const walk = (cats: CodexCategory[]) => {
   631→          for (const c of cats) {
   632→            if (c.entries.some((e) => e.id === entryId)) {
   633→              found = c;
   634→              return;
   635→            }
   636→            walk(c.children);
   637→          }
   638→        };
   639→        walk(novel.codex);
   640→        return found;
   641→      },
   642→    }),
   643→    {
   644→      name: STORAGE_KEY,
   645→      version: 3,
   646→      storage: createJSONStorage(() => localStorage),
   647→      /* v0 → v1: original-style statuses, free-text labels instead of label ids
   648→         v2 → v3: every note gets its icon + colour (like the original app —
   649→         acts and chapters as solid coloured folders) */
   650→      migrate: (persisted) => {
   651→        const s = persisted as KachamakState;
   652→        const OLD_DEFAULT_STATUSES = [
   653→          "To Do",
   654→          "First Draft",
   655→          "Second Draft",
   656→          "Revised",
   657→          "Final",
   658→        ];
   659→        const RENAMES: Record<string, string> = {
   660→          "Second Draft": "Revised Draft",
   661→          Revised: "Revised Draft",
   662→          Final: "Final Draft",
   663→        };
   664→        for (const n of s.novels ?? []) {
   665→          let upgraded = false;
   666→          if ((n.statuses ?? []).join("|") === OLD_DEFAULT_STATUSES.join("|")) {
   667→            n.statuses = [...DEFAULT_STATUSES];
   668→            upgraded = true;
   669→          }
   670→          const idToName = new Map((n.labels ?? []).map((l) => [l.id, l.name]));
   671→          const nameToColor = new Map((n.labels ?? []).map((l) => [l.name, l.color]));
   672→          /* act palette like the original screenshot: gold / green / purple … */
   673→          const ACT_PALETTE = [
   674→            { act: "#e3c567", chapter: "#c69c6d" },
   675→            { act: "#8ab661", chapter: "#699bdd" },
   676→            { act: "#a371f7", chapter: "#c084fc" },
   677→          ];
   678→          let actIndex = -1;
   679→          const walk = (nodes: NoteNode[], chapterColor?: string) => {
   680→            for (const nd of nodes) {
   681→              if (nd.label && idToName.has(nd.label)) nd.label = idToName.get(nd.label)!;
   682→              if (!nd.color && nd.label && nameToColor.has(nd.label)) {
   683→                nd.color = nameToColor.get(nd.label)!;
   684→              }
   685→              if (upgraded && RENAMES[nd.status]) nd.status = RENAMES[nd.status];
   686→              /* give every container-like note the solid folder icon + colours */
   687→              if (!nd.icon && (nd.kind === "act" || nd.kind === "chapter" || nd.kind === "folder")) {
   688→                nd.icon = "folder";
   689→              }
   690→              let chColor: string | undefined;
   691→              if (!nd.color && nd.kind === "act") {
   692→                actIndex += 1;
   693→                const pal = ACT_PALETTE[actIndex % ACT_PALETTE.length];
   694→                nd.color = pal.act;
   695→                chColor = pal.chapter;
   696→              }
   697→              if (!nd.color && nd.kind === "chapter" && chapterColor) {
   698→                nd.color = chapterColor;
   699→              }
   700→              walk(nd.children, nd.kind === "act" ? chColor : undefined);
   701→            }
   702→          };
   703→          for (const roots of Object.values(n.sections ?? {})) {
   704→            actIndex = -1;
   705→            walk(roots);
   706→          }
   707→          for (const t of n.trash ?? []) walk([t.node]);
   708→        }
   709→        return s;
   710→      },
   711→      onRehydrateStorage: () => (state) => {
   712→        // first run: seed with the demo novels
   713→        if (state && state.novels.length === 0) {
   714→          state.novels = seedNovels();
   715→        }
   716→      },
   717→    }
   718→  )
   719→);
