   134→interface LegacyCodexCategory {
   135→  id: string;
   136→  name: string;
   137→  children: LegacyCodexCategory[];
   138→  entries: LegacyCodexEntry[];
   139→}

   141→type LegacyNovel = Novel & { codex?: LegacyCodexCategory[] };
   142→
   143→const LEGACY_CAT_ICONS: [RegExp, string][] = [
   144→  [/^char/i, "users"],
   145→  [/^loc/i, "map-pin"],
   146→  [/^thing/i, "diamond"],
   147→  [/^lore/i, "scroll"],
   148→];

   150→/* turn an old codex category (with its entries) into a plain note subtree */
   151→function legacyCatToNode(cat: LegacyCodexCategory): NoteNode {
   152→  const now = Date.now();
   153→  const icon = LEGACY_CAT_ICONS.find(([re]) => re.test(cat.name))?.[1] ?? "folder";
   154→  return {
   155→    id: uid(),
   156→    title: cat.name,
   157→    kind: "folder",
   158→    synopsis: "",
   159→    content: "",
   160→    notes: "",
   161→    status: "",
   162→    label: "",
   163→    icon,
   164→    createdAt: now,
   165→    updatedAt: now,
   166→    children: [
   167→      ...cat.children.map(legacyCatToNode),
   168→      ...cat.entries.map((e) => ({
   169→        id: uid(),
   170→        title: e.title,
   171→        kind: "doc" as const,
   172→        synopsis: "",
   173→        content: e.content ?? "",
   174→        notes: "",
   175→        status: e.status ?? "",
   176→        label: e.entryType ?? "",
   177→        createdAt: e.createdAt ?? now,
   178→        updatedAt: e.updatedAt ?? now,
   179→        children: [] as NoteNode[],
   180→      })),
   181→    ],
   182→  };
   183→}
   184→
   185→/* make sure a novel has every section (older backups may miss keys) and convert
   186→   its legacy codex array into plain notes in the Codex section */
   187→function normalizeNovelSections(n: LegacyNovel): void {
   188→  const secs: Record<string, NoteNode[] | undefined> = { ...(n.sections ?? {}) };
   189→  for (const k of SECTION_KEYS) {
   190→    if (!Array.isArray(secs[k])) secs[k] = [];
   191→  }
   192→  if (secs.codex!.length === 0 && Array.isArray(n.codex) && n.codex.length > 0) {
   193→    secs.codex = n.codex.map(legacyCatToNode);
   194→  }
   195→  n.sections = secs as Novel["sections"];
   196→  delete n.codex;
   197→}

   199→/* ---------- immutable helpers ---------- */
   200→
   201→function mapNodes(nodes: NoteNode[], fn: (n: NoteNode) => NoteNode): NoteNode[] {
   202→  return nodes.map((n) => ({ ...fn(n), children: mapNodes(n.children, fn) }));
   203→}
   204→
   205→function removeFromNodes(
   206→  nodes: NoteNode[],
   207→  id: string
   208→): { nodes: NoteNode[]; removed: NoteNode | null; parentId: string | null } {
   209→  const found: { removed: NoteNode | null; parentId: string | null } = {
   210→    removed: null,
   211→    parentId: null,
   212→  };
   213→  const walk = (list: NoteNode[], pid: string | null): NoteNode[] => {
   214→    const out: NoteNode[] = [];
   215→    for (const n of list) {
   216→      if (n.id === id) {
   217→        found.removed = n;
   218→        found.parentId = pid;
   219→        continue;
   220→      }
   221→      out.push({ ...n, children: walk(n.children, n.id) });
   222→    }
   223→    return out;
   224→  };
   225→  const nodes2 = walk(nodes, null);
   226→  return { nodes: nodes2, removed: found.removed, parentId: found.parentId };
   227→}

   229→function touch(novel: Novel): Novel {
   230→  return { ...novel, updatedAt: Date.now() };
   231→}

   242→    status: "",
   243→    label: "",
   244→    createdAt: now,
   245→    updatedAt: now,
   246→    children: [],
   247→  };
   248→}

   504→      deleteNode: (section, id) => {

   509→          id
   510→        );

   512→        novel.sections = { ...novel.sections, [section]: nodes };
   513→        novel.trash = [
   514→          { id: uid(), parentId, section, node: removed, deletedAt: Date.now() },
   515→          ...novel.trash,

   520→            !!nid && (nid === id || !!findNode([removedNode], nid));
   521→          return {
   522→            selectedNodeId: gone(s.selectedNodeId) ? null : s.selectedNodeId,
   523→            paneAId: gone(s.paneAId) ? null : s.paneAId,
   524→            paneBId: gone(s.paneBId) ? null : s.paneBId,
   525→            renamingNodeId:
   526→              s.renamingNodeId &&
   527→              (s.renamingNodeId === id || findNode([removedNode], s.renamingNodeId))
   528→                ? null
   529→                : s.renamingNodeId,
   530→            novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   531→          };
   532→        });
   533→      },
   534→
   535→      moveNode: (section, id, dir) => {
   536→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   537→        if (!novel) return;
   538→        const roots = novel.sections[section];
   539→
   540→        const swap = (list: NoteNode[]): boolean => {
   541→          const i = list.findIndex((n) => n.id === id);
   542→          if (i >= 0) {
   543→            const j = dir === "up" ? i - 1 : i + 1;
   544→            if (j < 0 || j >= list.length) return false;
   545→            const copy = [...list];
   546→            [copy[i], copy[j]] = [copy[j], copy[i]];
   547→            list.length = 0;
   548→            list.push(...copy);
   549→            return true;
   550→          }
   551→          return list.some((n) => swap(n.children));
   552→        };
   553→        swap(roots);
   554→        set((s) => ({
   555→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   556→        }));
   557→      },
   558→
   559→      /* Move — the note becomes the last child of the destination (or a section root) */
   560→      moveNodeTo: (fromSection, id, destSection, destParentId, destIndex) => {
   561→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   562→        if (!novel) return;
   563→        const fromRoots = novel.sections[fromSection];
   564→        const moving = findNode(fromRoots, id);
   565→        if (!moving) return;
   566→        /* never move a note into itself or one of its own sub-pages */
   567→        if (
   568→          destSection === fromSection &&
   569→          destParentId &&
   570→          (destParentId === id || findNode([moving], destParentId))
   571→        )
   572→          return;
   573→        const { nodes, removed } = removeFromNodes(fromRoots, id);
   574→        if (!removed) return;
   575→        novel.sections = { ...novel.sections, [fromSection]: nodes };
   576→        /* destIndex (optional): insertion position in the post-removal list —
   577→           the Corkboard computes it against the array without the dragged note */
   578→        const spliceOrPush = (list: NoteNode[]) => {
   579→          if (typeof destIndex === "number") {
   580→            const idx = Math.max(0, Math.min(destIndex, list.length));
   581→            list.splice(idx, 0, removed);
   582→          } else {
   583→            list.push(removed);
   584→          }
   585→        };
   586→        if (destParentId) {
   587→          const parent = findNode(novel.sections[destSection], destParentId);
   588→          if (parent) {
   589→            spliceOrPush(parent.children);
   590→            parent.collapsed = false;
   591→          } else {
   592→            spliceOrPush(novel.sections[destSection]);
   593→          }
   594→        } else {
   595→          spliceOrPush(novel.sections[destSection]);
   596→        }
   597→        set((s) => ({
   598→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   599→        }));
   600→      },
   601→
   602→      /* deep-copy the note (with all sub-pages) right below the original */
   603→      duplicateNode: (section, id) => {
   604→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   605→        if (!novel) return;
   606→        const roots = novel.sections[section];
   607→        if (!findNode(roots, id)) return;
   608→        const deep = (n: NoteNode): NoteNode => ({
   609→          ...n,
   610→          id: uid(),
   611→          createdAt: Date.now(),
   612→          updatedAt: Date.now(),
   613→          children: n.children.map(deep),
   614→        });
   615→        const insertAfter = (list: NoteNode[]): boolean => {
   616→          const i = list.findIndex((n) => n.id === id);
   617→          if (i >= 0) {
   618→            list.splice(i + 1, 0, deep(list[i]));
   619→            return true;
   620→          }
   621→          return list.some((n) => insertAfter(n.children));
   622→        };
   623→        insertAfter(roots);
   624→        set((s) => ({
   625→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   626→        }));
   627→      },
   628→
   629→      /* sort sub-pages of a note — title A→Z / Z→A, oldest or newest first */
   630→      sortChildren: (section, id, mode) => {
   631→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   632→        if (!novel) return;
   633→        const roots = novel.sections[section];
   634→        const target = findNode(roots, id);
   635→        const list = target ? target.children : roots;
   636→        if (!target || list.length === 0) return;
   637→        const byTitle = (a: NoteNode, b: NoteNode) =>
   638→          a.title.localeCompare(b.title, undefined, {
   639→            sensitivity: "base",
   640→            numeric: true,
   641→          });
   642→        list.sort((a, b) => {
   643→          switch (mode) {
   644→            case "title-desc":
   645→              return byTitle(b, a);
   646→            case "created-asc":
   647→              return a.createdAt - b.createdAt;
   648→            case "created-desc":
   649→              return b.createdAt - a.createdAt;
   650→            default:
   651→              return byTitle(a, b);
   652→          }
   653→        });
   654→        set((s) => ({
   655→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   656→        }));
   657→      },
   658→
   659→      /* ---------- trash ---------- */
   660→
   661→      restoreFromTrash: (trashId) => {
   662→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   663→        if (!novel) return;
   664→        const item = novel.trash.find((t) => t.id === trashId);
   665→        if (!item) return;
   666→        if (item.parentId) {
   667→          const parent = findNode(novel.sections[item.section], item.parentId);
   668→          if (parent) {
   669→            parent.children.push(item.node);
   670→            parent.collapsed = false;
   671→          } else {
   672→            novel.sections[item.section].push(item.node);
   673→          }
   674→        } else {
   675→          novel.sections[item.section].push(item.node);
   676→        }
   677→        novel.trash = novel.trash.filter((t) => t.id !== trashId);
   678→        set((s) => ({
   679→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   680→        }));
   681→      },
   682→
   683→      deleteForever: (trashId) => {
   684→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   685→        if (!novel) return;
   686→        novel.trash = novel.trash.filter((t) => t.id !== trashId);
   687→        set((s) => ({
   688→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   689→        }));
   690→      },
   691→
   692→      emptyTrash: () => {
   693→        const novel = get().novels.find((n) => n.id === get().activeNovelId);
   694→        if (!novel) return;
   695→        novel.trash = [];
   696→        set((s) => ({
   697→          novels: s.novels.map((x) => (x.id === novel.id ? touch({ ...x }) : x)),
   698→        }));
   699→      },
   700→    }),
   701→    {
   702→      name: STORAGE_KEY,
   703→      version: 5,
   704→      storage: createJSONStorage(() => localStorage),
   705→      /* v0 → v1: original-style statuses, free-text labels instead of label ids
   706→         v2 → v3: every note gets its icon + colour (like the original app —
   707→         acts and chapters as solid coloured folders)
   708→         v3 → v4: everything is a NOTE on every tab — the old Codex
   709→         categories/entries become plain notes in the Codex section
   710→         v4 → v5: the Grid view becomes the Corkboard (Scrivener index cards) */
   711→      migrate: (persisted) => {
   712→        const s = persisted as KachamakState;
   713→        if ((s as { view?: string }).view === "grid") s.view = "corkboard";
   714→        const OLD_DEFAULT_STATUSES = [
   715→          "To Do",
   716→          "First Draft",
   717→          "Second Draft",
   718→          "Revised",
   719→          "Final",
   720→        ];
   721→        const RENAMES: Record<string, string> = {
   722→          "Second Draft": "Revised Draft",
   723→          Revised: "Revised Draft",
   724→          Final: "Final Draft",
   725→        };
   726→        for (const n of (s.novels ?? []) as LegacyNovel[]) {
   727→          let upgraded = false;
   728→          if ((n.statuses ?? []).join("|") === OLD_DEFAULT_STATUSES.join("|")) {
   729→            n.statuses = [...DEFAULT_STATUSES];
   730→            upgraded = true;
   731→          }
   732→          const idToName = new Map((n.labels ?? []).map((l) => [l.id, l.name]));
   733→          const nameToColor = new Map((n.labels ?? []).map((l) => [l.name, l.color]));
   734→          /* act palette like the original screenshot: gold / green / purple … */
   735→          const ACT_PALETTE = [
   736→            { act: "#e3c567", chapter: "#c69c6d" },
   737→            { act: "#8ab661", chapter: "#699bdd" },
   738→            { act: "#a371f7", chapter: "#c084fc" },
   739→          ];
   740→          let actIndex = -1;
   741→          const walk = (nodes: NoteNode[], chapterColor?: string) => {
   742→            for (const nd of nodes) {
   743→              if (nd.label && idToName.has(nd.label)) nd.label = idToName.get(nd.label)!;
   744→              if (!nd.color && nd.label && nameToColor.has(nd.label)) {
   745→                nd.color = nameToColor.get(nd.label)!;
   746→              }
   747→              if (upgraded && RENAMES[nd.status]) nd.status = RENAMES[nd.status];
   748→              /* give every container-like note the solid folder icon + colours */
   749→              if (!nd.icon && (nd.kind === "act" || nd.kind === "chapter" || nd.kind === "folder")) {
   750→                nd.icon = "folder";
   751→              }
   752→              let chColor: string | undefined;
   753→              if (!nd.color && nd.kind === "act") {
   754→                actIndex += 1;
   755→                const pal = ACT_PALETTE[actIndex % ACT_PALETTE.length];
   756→                nd.color = pal.act;
   757→                chColor = pal.chapter;
   758→              }
   759→              if (!nd.color && nd.kind === "chapter" && chapterColor) {
   760→                nd.color = chapterColor;
   761→              }
   762→              walk(nd.children, nd.kind === "act" ? chColor : undefined);
   763→            }
   764→          };
   765→          normalizeNovelSections(n);
   766→          for (const roots of Object.values(n.sections)) {
   767→            actIndex = -1;
   768→            walk(roots);
   769→          }
   770→          for (const t of n.trash ?? []) walk([t.node]);
   771→        }
   772→        return s;
   773→      },
   774→      onRehydrateStorage: () => (state) => {
   775→        // first run: seed with the demo novels
   776→        if (state && state.novels.length === 0) {
   777→          state.novels = seedNovels();
   778→        }
   779→      },
   780→    }
   781→  )
   782→);
